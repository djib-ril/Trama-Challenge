const socket = io();


// JOIN GAME

function joinGame(){

  const name =
    document.getElementById("username")
    .value
    .trim();

  if(!name){
    return alert("Enter nickname");
  }

  socket.emit("joinGame", name);

  document.getElementById("username")
    .disabled = true;

  document.getElementById("startBtn")
    .classList.remove("hidden");
}


// START GAME

function startGame(){
  socket.emit("startGame");
}


// SUBMIT SUMMARY

function submitSummary(){

  const text =
    document.getElementById("summaryInput")
    .value
    .trim();

  if(!text){
    return alert("Write a summary first");
  }

  socket.emit("submitSummary", text);

  document.getElementById("submitBtn")
    .disabled = true;

  document.getElementById("summaryInput")
    .disabled = true;
}


// UPDATE PLAYERS

socket.on("updatePlayers", (players)=>{

  const list =
    document.getElementById("playerList");

  list.innerHTML =
    players.map(
      p => "<li>" + p + "</li>"
    ).join("");

});


// GAME STARTED

socket.on("gameStarted", (passage)=>{

  document.getElementById("lobbyScreen")

  document.getElementById("gameScreen")
    .classList.remove("hidden");

  document.getElementById("passageBox")
    .innerText = passage;

});

  document.getElementById("timer")

  if(timeLeft <= 0){

    document.getElementById("summaryInput")
      .disabled = true;

    document.getElementById("submitBtn")

});

// ================= GAME STATE =================

let gameState = {

  passage:
    "Artificial Intelligence is transforming society through healthcare, logistics, automation, and technology innovation. Ethical concerns like privacy, bias, and regulation must be carefully managed.",

  players: {},

  timer: 120,

  isGameActive: false
};


// ================= SOCKET.IO =================

io.on("connection", (socket) => {

  console.log("Player connected");


  // JOIN

  socket.on("joinGame", (username) => {

    gameState.players[socket.id] = {
      name: username,
      score: 0,
      summary: ""
    };

    io.emit(
      "updatePlayers",
      Object.values(gameState.players)
        .map(p => p.name)
    );

  });


  // START GAME

  socket.on("startGame", () => {

    if(!gameState.isGameActive){

      gameState.isGameActive = true;

      io.emit(
        "gameStarted",
        gameState.passage
      );

      startCountdown();
    }

  });


  // SUBMIT SUMMARY

  socket.on("submitSummary", (summary) => {

    if(!gameState.players[socket.id]){
      return;
    }

    gameState.players[socket.id]
      .summary = summary;

    const keywords = [
      "artificial",
      "healthcare",
      "automation",
      "privacy",
      "bias",
      "regulation"
    ];

    let score = 0;

    const cleanSummary =
      summary.toLowerCase();

    keywords.forEach((word)=>{

      if(cleanSummary.includes(word)){
        score += 25;
      }

    });

    const wordCount =
      summary
      .split(/\\s+/)
      .filter(Boolean)
      .length;

    if(wordCount >= 15 && wordCount <= 40){
      score += 50;
    }

    gameState.players[socket.id]
      .score = score;

  });


  // DISCONNECT

  socket.on("disconnect", ()=>{

    delete gameState.players[socket.id];

    io.emit(
      "updatePlayers",
      Object.values(gameState.players)
        .map(p => p.name)
    );

  });

});


// ================= TIMER =================

function startCountdown(){

  let timeLeft = gameState.timer;

  const interval = setInterval(()=>{

    timeLeft--;

    io.emit("timerUpdate", timeLeft);

    if(timeLeft <= 0){

      clearInterval(interval);

      gameState.isGameActive = false;

      const leaderboard =
        Object.values(gameState.players)
        .sort((a,b)=> b.score - a.score);

      io.emit("gameOver", leaderboard);

    }

  },1000);

}


// ================= START SERVER =================
